PREFIX dct: <http://purl.org/dc/terms/>
PREFIX notes: <https://notes.example/notes/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
# Titles of the notes created this week
SELECT DISTINCT ?note ?title WHERE {
  GRAPH ?g { ?note a notes:Note ; dct:title ?title ; dct:created ?d . }
  FILTER (?d >= "2026-10-20"^^xsd:date && LANG(?title) = "en")
  OPTIONAL { ?note dct:creator _:who }
}
ORDER BY DESC(?d) LIMIT 500
